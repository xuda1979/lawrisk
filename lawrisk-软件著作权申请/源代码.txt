"""Shared text utilities: paragraph extraction, normalisation, heading detection."""
from __future__ import annotations
import re
from typing import List
_WS_RE = re.compile(r"\s+")
HEADING_RE = re.compile(
    r"^\s*"
    r"(?:"
    r"ARTICLE[.\-\s]+[0-9IVXLC]+|"
    r"SECTION[.\-\s]+[0-9IVXLC]+|"
    r"\d+\.\d*\s+[A-Z]|"
    r"第\s*[一二三四五六七八九十百\d]+\s*(?:条|章)"
    r")",
    re.IGNORECASE,
)
def normalize(text: str) -> str:
    """Canonical normalisation used for deterministic compare tokens."""
    if text is None:
        return ""
    t = text.casefold()
    t = _WS_RE.sub(" ", t)
    return t.strip()
def split_paragraphs(text: str) -> List[str]:
    """Split raw document text into non-empty paragraphs."""
    paras = [p.strip() for p in re.split(r"\n\s*\n|\r\n\r\n", text)]
    return [p for p in paras if p]
def looks_like_heading(text: str, prev: str | None = None) -> bool:
    """Heuristics: numbered section/title patterns, or short lines before a blank."""
    t = text.strip()
    if not t:
        return False
    if len(t) > 80:
        return False
    if HEADING_RE.match(t):
        return True
    if re.match(r"^[A-Z0-9][A-Z0-9 _/:&-]{1,40}$", t) and len(t) <= 60:
        return True
    return False

def _read_text_from_docx(data: bytes) -> str:
    import docx
    with tempfile.NamedTemporaryFile(suffix=".docx", delete=False) as tmp:
        tmp.write(data)
        tmp_path = tmp.name
    try:
        document = docx.Document(tmp_path)
        blocks: List[str] = []
        last_empty = True
        for para in document.paragraphs:
            t = para.text.strip()
            if t:
                blocks.append(t)
                last_empty = False
            elif not last_empty:
                blocks.append("")
                last_empty = True
        text = "\n\n".join(blocks)
        for table in document.tables:
            for row in table.rows:
                row_txt = " | ".join(c.text.strip() for c in row.cells if c.text.strip())
                if row_txt:
                    text += "\n\n" + row_txt
        return text
    finally:
        Path(tmp_path).unlink(missing_ok=True)

def _risk_by_para(report: RiskReport) -> Dict[int, List[dict]]:
    """Map global paragraph index -> penalties that fired on that paragraph."""
    out: Dict[int, List[dict]] = {}
    for p in report.penalties:
        if p.para_index is None:
            continue
        out.setdefault(p.para_index, []).append(
            {
                "rule_id": p.rule_id,
                "dimension": p.dimension,
                "severity": p.severity,
                "reason": p.reason,
                "mitigation": p.mitigation,
            }
        )
    return out
def _attach_row_risks(rows: List[dict], risk_a: RiskReport, risk_b: RiskReport) -> List[dict]:
    """Attribute risk penalties to diff rows.
    Diff rows and the risk engine index the same global paragraph arrays, so a
    penalty's para_index tells us exactly which changed clause carried the
    points. Each row gains per-side lists (leftRisks/rightRisks) plus the
    summed severity so the web diff can flag risky changed clauses.
    """
    left_map = _risk_by_para(risk_a)
    right_map = _risk_by_para(risk_b)
    for r in rows:
        lk = r.get("leftIdx")
        rk = r.get("rightIdx")
        left = left_map.get(lk, []) if lk is not None else []
        right = right_map.get(rk, []) if rk is not None else []
        r["leftRisks"] = left
        r["rightRisks"] = right
        r["leftRiskSum"] = sum(p["severity"] for p in left)
        r["rightRiskSum"] = sum(p["severity"] for p in right)
    return rows

def word_diff(left: str, right: str) -> Tuple[List[str], List[str]]:
    """Token-level diff for inline highlight; returns (removed, added)."""
    l_tokens = _WORD_RE.findall(left) if left else []
    r_tokens = _WORD_RE.findall(right) if right else []
    sm = difflib.SequenceMatcher(a=l_tokens, b=r_tokens, autojunk=False)
    removed: List[str] = []
    added: List[str] = []
    for tag, i1, i2, j1, j2 in sm.get_opcodes():
        if tag in ("delete", "replace"):
            removed.extend(l_tokens[i1:i2])
        if tag in ("insert", "replace"):
            added.extend(r_tokens[j1:j2])
    return removed, added
def _pair_replace(
    left: Sequence[str], right: Sequence[str],
) -> Tuple[List[Tuple[str, str]], List[str], List[str]]:
    """Align a replace block pair-wise; returns (pairs, extra_left, extra_right)."""
    pairs: List[Tuple[str, str]] = []
    for i, j in zip(range(len(left)), range(len(right))):
        pairs.append((left[i], right[j]))
    extra_left = list(left[len(right):]) if len(left) > len(right) else []
    extra_right = list(right[len(left):]) if len(right) > len(left) else []
    return pairs, extra_left, extra_right

def _top_risky_clauses(sections, left_map, right_map, n: int = 5):
    """Rank changed clauses by combined A/B risk severity; return top n.
    Each entry is ``(section_label, DiffItem, total_severity)`` ordered by
    descending total points. Clauses with no fired penalties are excluded.
    """
    ranked = []
    for label, items in sections:
        for it in items:
            left = left_map.get(it.para_index_left, []) if it.para_index_left is not None else []
            right = right_map.get(it.para_index_right, []) if it.para_index_right is not None else []
            total = sum(r["severity"] for r in left) + sum(r["severity"] for r in right)
            if total:
                ranked.append((label, it, total))
    ranked.sort(key=lambda t: t[2], reverse=True)
    return ranked[:n]
def _risk_by_section(sections, left_map, right_map):
    """Per-section A/B combined risk points, for the review-triage table.
    Returns ``(label, left_pts, right_pts, combined)`` for sections that carry
    at least one fired penalty, ordered by descending combined points.
    """
    rows = []
    for label, items in sections:
        left_pts = right_pts = 0
        for it in items:
            left = left_map.get(it.para_index_left, []) if it.para_index_left is not None else []
            right = right_map.get(it.para_index_right, []) if it.para_index_right is not None else []
            left_pts += sum(r["severity"] for r in left)
            right_pts += sum(r["severity"] for r in right)
        if left_pts or right_pts:
            rows.append((label, left_pts, right_pts, left_pts + right_pts))
    rows.sort(key=lambda t: t[3], reverse=True)
    return rows

        name = DIM_LABEL.get(str(dim), str(dim))

"""Optional LLM enhancement layer (P3).
Design
------
* OFF by default. Unless the environment variable ``LAWRISK_LLM=1`` is set, all
  summarisation routes use the deterministic :class:`MockSummarizer` so the
  whole pipeline stays local, offline and reproducible.
* When enabled, :func:`get_provider` returns a pluggable provider built via the
  ``LAWRISK_LLM_PROVIDER`` env var (built-ins: ``mock``; extend by subclassing
  :class:`Summarizer`). A real implementation would call an HTTP endpoint; the
  interface stays a pure function of ``(left, right, op)`` so callers never
  touch the network themselves.
Both providers are deterministic for identical inputs, preserving AC1.
"""

import re
from typing import List
ENABLED = os.environ.get("LAWRISK_LLM", "0").lower() in {"1", "true", "yes"}
PROVIDER_NAME = os.environ.get("LAWRISK_LLM_PROVIDER", "mock").lower()
class Summarizer:
    """Pluggable summarisation provider interface."""
    name = "base"
    def summarize_clause(
        self, text_left: str | None, text_right: str | None, op: str
    ) -> str:
        """Return a one-line semantic summary for one diff item."""
        raise NotImplementedError
    def summarize_report(self, stats, counts: List[str]) -> str:
        """Return a paragraph-level summary for the whole report."""
        raise NotImplementedError
def _preview(t: str | None, n: int = 90) -> str:
    if not t:
        return ""
    clean = re.sub(r"\s+", " ", t).strip()
    return clean if len(clean) <= n else clean[: n - 1] + "…"

class _ProviderRegistry:
    def __init__(self) -> None:
        self._providers: dict[str, type[Summarizer]] = {
            MockSummarizer.name: MockSummarizer,
        }
    def register(self, cls: type[Summarizer]) -> None:
        self._providers[cls.name] = cls
    def build(self, name: str) -> Summarizer:
        cls = self._providers.get(name or "mock")
        if cls is None:
            raise ValueError(f"unknown LLM provider: {name!r}")
        return cls()
registry = _ProviderRegistry()
def get_provider(name: str | None = None) -> Summarizer:
    """Build the active summariser from the registry.
    An explicit ``name`` always wins (used by tests to force a provider). With

    stays stable when the layer is off.

    if name is not None:
        return registry.build(name)
    requested = PROVIDER_NAME if ENABLED else "mock"
    return registry.build(requested)
